Silver: type without raising, one rule column

sugars   = F.expr(r"try_cast(regexp_replace(trim(sugars_100g), '\\s*g$', '') AS double)")
modified = F.coalesce(F.to_date(F.try_to_timestamp("last_modified", F.lit("yyyy-MM-dd"))),
                      F.to_date(F.try_to_timestamp("last_modified", F.lit("dd/MM/yyyy"))))
rule = (F.when(F.col("code").isNull() | (F.trim("code") == ""), "code_missing")
         .when(F.col("_modified").isNull(), "last_modified_unparseable")
         .when(F.col("sugars_100g").isNotNull() & F.col("_sugars").isNull(), "sugars_not_a_number")
         .when((F.col("_sugars") < 0) | (F.col("_sugars") > 100), "sugars_out_of_range"))
# good = rule IS NULL; rejected = rule IS NOT NULL (+ rejected_at) -> quarantine, mode("overwrite")

Silver: one row per key, latest wins

SILVER_COLUMNS = ["code", "product_name", "brands", "categories", "countries", "nutrition_grade_fr", "sugars_100g", "last_modified"]   # lesson 4's good, minus loaded_at
w = Window.partitionBy("code").orderBy(F.desc("last_modified"), F.desc("loaded_at"))
latest = good.withColumn("_n", F.row_number().over(w)).filter("_n = 1").select(*SILVER_COLUMNS)

if not DeltaTable.isDeltaTable(spark, silver):
    latest.limit(0).write.format("delta").save(silver)
(DeltaTable.forPath(spark, silver).alias("t")
   .merge(latest.alias("s"), "t.code = s.code")
   .whenMatchedUpdateAll(condition="s.last_modified > t.last_modified")
   .whenNotMatchedInsertAll().execute())

Gold: reshape, never repair

(s.withColumn("category", F.explode(F.split("categories", ",")))
  .withColumn("country", F.explode(F.split("countries", ",")))
  .groupBy("category", "country", "nutrition_grade_fr").agg(F.count("*").alias("products"))
  .write.format("delta").mode("overwrite").save(f"{gold}/nutriscore_by_category"))

History, time travel, restore

dt = DeltaTable.forPath(spark, path)
dt.history().select("version", "operation", "operationMetrics.numTargetRowsInserted").show()
spark.read.format("delta").option("versionAsOf", 3).load(path)
dt.restoreToVersion(3)          # a new commit; nothing erased until VACUUM (7 days retention by default)
